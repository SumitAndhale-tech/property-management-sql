-- ============================================================
-- Q1. Find the number of units in each property.
-- Concept: JOIN + GROUP BY + COUNT
-- ============================================================

SELECT
    p.property_name,
    COUNT(u.unit_id) AS total_units
FROM properties p
INNER JOIN units u
    ON p.property_id = u.property_id
GROUP BY p.property_id, p.property_name;


-- ============================================================
-- Q2. Find all vacant units.
-- Concept: WHERE
-- ============================================================

SELECT
    unit_id,
    property_id,
    unit_number,
    unit_type,
    monthly_rent
FROM units
WHERE status = 'Vacant';


-- ============================================================
-- Q3. Find units with monthly rent greater than ₹25,000.
-- Concept: WHERE + comparison operator
-- ============================================================

SELECT
    unit_id,
    unit_number,
    unit_type,
    monthly_rent
FROM units
WHERE monthly_rent > 25000;


-- ============================================================
-- Q4. Find the top 5 most expensive units.
-- Concept: ORDER BY + LIMIT
-- ============================================================

SELECT
    unit_id,
    unit_number,
    unit_type,
    monthly_rent
FROM units
ORDER BY monthly_rent DESC
LIMIT 5;

-- Q25. Find the property with the highest total paid rental
--      revenue.
-- Concept: CTE + GROUP BY + SUM + Subquery
-- ============================================================

WITH property_revenue AS (
    SELECT
        p.property_id,
        p.property_name,
        SUM(pay.amount) AS total_revenue
    FROM properties p
    INNER JOIN units u
        ON p.property_id = u.property_id
    INNER JOIN leases l
        ON u.unit_id = l.unit_id
    INNER JOIN payments pay
        ON l.lease_id = pay.lease_id
    WHERE pay.payment_status = 'Paid'
    GROUP BY p.property_id, p.property_name
)
SELECT *
FROM property_revenue
WHERE total_revenue = (
    SELECT MAX(total_revenue)
    FROM property_revenue
);

-- ============================================================
-- Q27. Find the property/properties with the highest occupancy
--      rate.
-- Concept: CTE + CASE + Calculation + Subquery
-- ============================================================

WITH property_occupancy AS (
    SELECT
        p.property_id,
        p.property_name,
        COUNT(u.unit_id) AS total_units,
        SUM(
            CASE
                WHEN u.status = 'Occupied' THEN 1
                ELSE 0
            END
        ) AS occupied_units
    FROM properties p
    INNER JOIN units u
        ON p.property_id = u.property_id
    GROUP BY p.property_id, p.property_name
)
SELECT
    property_name,
    total_units,
    occupied_units,
    (occupied_units * 100.0 / total_units) AS occupancy_rate
FROM property_occupancy
WHERE (occupied_units * 100.0 / total_units) = (
    SELECT MAX(occupied_units * 100.0 / total_units)
    FROM property_occupancy
);

-- ============================================================
-- Q29. Find each property's paid rental revenue, paid expenses,
--      and net amount.
-- Concept: Multiple CTEs + LEFT JOIN + COALESCE
-- ============================================================

WITH property_revenue AS (
    SELECT
        p.property_id,
        p.property_name,
        SUM(pay.amount) AS total_revenue
    FROM properties p
    INNER JOIN units u
        ON p.property_id = u.property_id
    INNER JOIN leases l
        ON u.unit_id = l.unit_id
    INNER JOIN payments pay
        ON l.lease_id = pay.lease_id
    WHERE pay.payment_status = 'Paid'
    GROUP BY p.property_id, p.property_name
),
property_expenses AS (
    SELECT
        p.property_id,
        SUM(e.amount) AS total_expenses
    FROM properties p
    INNER JOIN expenses e
        ON p.property_id = e.property_id
    WHERE e.payment_status = 'Paid'
    GROUP BY p.property_id
)
SELECT
    r.property_name,
    r.total_revenue,
    COALESCE(e.total_expenses, 0) AS total_expenses,
    r.total_revenue - COALESCE(e.total_expenses, 0) AS net_amount
FROM property_revenue r
LEFT JOIN property_expenses e
    ON r.property_id = e.property_id;


-- ============================================================
-- Q30. Find properties having occupancy above 80% and total
--      paid rental revenue above ₹100,000.
-- Concept: CTE + CASE + JOINs + GROUP BY + HAVING/WHERE
-- ============================================================

WITH property_analysis AS (
    SELECT
        p.property_id,
        p.property_name,
        COUNT(DISTINCT u.unit_id) AS total_units,
        COUNT(
            DISTINCT CASE
                WHEN u.status = 'Occupied' THEN u.unit_id
            END
        ) AS occupied_units,
        SUM(
            CASE
                WHEN pay.payment_status = 'Paid'
                THEN pay.amount
                ELSE 0
            END
        ) AS total_revenue
    FROM properties p
    INNER JOIN units u
        ON p.property_id = u.property_id
    LEFT JOIN leases l
        ON u.unit_id = l.unit_id
    LEFT JOIN payments pay
        ON l.lease_id = pay.lease_id
    GROUP BY p.property_id, p.property_name
)
SELECT
    property_name,
    (occupied_units * 100.0 / total_units) AS occupancy_rate,
    total_revenue
FROM property_analysis
WHERE (occupied_units * 100.0 / total_units) > 80
  AND total_revenue > 100000;
